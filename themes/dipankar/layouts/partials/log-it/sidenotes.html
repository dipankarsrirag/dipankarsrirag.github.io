{{- /* Copies each footnote ([^1] in the text) next to its reference as a sidenote, so wide screens can show it in the right margin. The endnote list stays in the HTML for narrow screens. Input: the post's HTML. */ -}} {{- $html := . -}} {{- $notes := dict -}} {{- range findRESubmatch `
  • \s*([\s\S]*?)
    • ` $html -}} {{- $body := index . 2 -}} {{- $body = replaceRE `( )?]*>[^<]*` "" $body -}} {{- $body = replaceRE `

    \s*

    ` `` $body -}} {{- $body = replaceRE `` "" $body -}} {{- $notes = merge $notes (dict (index . 1) (trim $body " \n")) -}} {{- end -}} {{- range findRESubmatch `([^<]+)` $html -}} {{- $ref := index . 0 -}} {{- $num := index . 2 -}} {{- with index $notes (index . 1) -}} {{- $html = replace $html $ref (printf `%s%s %s` $ref $num .) -}} {{- end -}} {{- end -}} {{- $html | safeHTML -}}
