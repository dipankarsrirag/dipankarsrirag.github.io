{{- /* Research map for a log-it folder with `themes:` in its _index.md: themes (left) and papers (right), each paper once, joined by curves coloured by theme. Drawn at build time; a few lines of script add the hover highlight. Papers come from content/publications (`themes:` ids, optional `short:` name). A paper links to its log-it post once published, otherwise to the paper. Themes may set their own `color:`. */ -}} {{- $folder := . -}} {{- $palette := slice "#3e6e8e" "#554882" "#c0673f" "#5e8c61" "#b8892b" "#8e5b8a" "#4f8a8b" "#a24d5a" -}} {{- $themeDefs := slice }}{{ range $i, $t := $folder.Params.themes }}{{ $themeDefs = $themeDefs | append (merge $t (dict "i" $i "color" ($t.color | default (index $palette (mod $i (len $palette)))))) }}{{ end -}} {{- $tIndex := dict }}{{ range $themeDefs }}{{ $tIndex = merge $tIndex (dict .id .i) }}{{ end -}} {{- $posts := (site.GetPage "/log-it").RegularPagesRecursive -}} {{- $papers := slice -}} {{- range (where site.RegularPages "Section" "publications").ByDate -}} {{- $p := . -}} {{- $mine := slice }}{{ $sum := 0 -}} {{- range $themeDefs }}{{ if in ($p.Params.themes | default slice) .id }}{{ $mine = $mine | append .id }}{{ $sum = add $sum .i }}{{ end }}{{ end -}} {{- if $mine -}} {{- $key := path.BaseName $p.File.Path -}} {{- $url := $p.Params.links.paper | default "" }}{{ $external := true -}} {{- with index (where $posts "Params.paper" $key) 0 }}{{ $url = .RelPermalink }}{{ $external = false }}{{ end -}} {{- $venue := $p.Params.venue | default "" | replaceRE `\s*\d{4}\s*$` "" -}} {{- if $p.Params.preprint }}{{ $venue = "Preprint" }}{{ end -}} {{- $papers = $papers | append (dict "key" $key "themes" $mine "label" ($p.Params.short | default (index (split $p.Title ":") 0)) "title" $p.Title "meta" (printf "%s%s%s" $venue (cond (ne $venue "") " · " "") ($p.Date.Format "2006")) "url" $url "external" $external "preprint" ($p.Params.preprint | default false) "sort" (printf "%07.3f|%012d" (div (float $sum) (len $mine)) $p.Date.Unix)) -}} {{- end -}} {{- end -}} {{- $papers = sort $papers "sort" -}} {{- if $papers -}} {{- $used := slice }}{{ range $themeDefs }}{{ $t := . }}{{ $n := 0 }}{{ range $papers }}{{ if in .themes $t.id }}{{ $n = add $n 1 }}{{ end }}{{ end }}{{ if $n }}{{ $used = $used | append (merge . (dict "n" $n)) }}{{ end }}{{ end -}} {{- $color := dict }}{{ range $used }}{{ $color = merge $color (dict .id .color) }}{{ end -}} {{- $rowH := 44.0 }}{{ $pad := 20.0 -}} {{- $tDot := 12.0 }}{{ $tLabel := 34.0 -}} {{- /* Edges start just after the longest theme name (≈8.4 px per character at 15 px) */ -}} {{- $tLongest := 0 }}{{ range $used }}{{ if gt (len .name) $tLongest }}{{ $tLongest = len .name }}{{ end }}{{ end -}} {{- $edgeFrom := add $tLabel (mul $tLongest 8.4) 16 }}{{ $nodeX := add $edgeFrom 200 }}{{ $labelX := add $nodeX 16 -}} {{- $H := add (mul (len $papers) $rowH) (mul $pad 2) -}} {{- $longest := 0 }}{{ range $papers }}{{ if gt (len .label) $longest }}{{ $longest = len .label }}{{ end }}{{ end -}} {{- $W := add $labelX (mul $longest 8.2) 10 -}} {{- $py := dict }}{{ range $i, $p := $papers }}{{ $py = merge $py (dict $p.key (add $pad (add (mul $i $rowH) (div $rowH 2)))) }}{{ end -}} {{- $tGap := div (sub $H (mul $pad 2)) (len $used) -}} {{- $ty := dict }}{{ range $i, $t := $used }}{{ $ty = merge $ty (dict $t.id (add $pad (add (mul $i $tGap) (div $tGap 2)))) }}{{ end -}} {{- end -}}
